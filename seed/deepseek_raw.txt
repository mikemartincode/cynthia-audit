def compare_ref(pair):
    a, b = pair

    def parse_version(v):
        # strip build metadata
        v = v.split('+')[0]
        # split core and prerelease
        if '-' in v:
            core_str, pre_str = v.split('-', 1)
        else:
            core_str, pre_str = v, ''
        # parse core
        parts = core_str.split('.')
        if len(parts) != 3:
            raise ValueError('Invalid semver')
        major = int(parts[0])
        minor = int(parts[1])
        patch = int(parts[2])
        # prerelease identifiers
        pre_ids = pre_str.split('.') if pre_str else []
        return (major, minor, patch, pre_ids)

    def compare_ids(id1, id2):
        is_num1 = id1.isdigit()
        is_num2 = id2.isdigit()
        if is_num1 and is_num2:
            n1 = int(id1)
            n2 = int(id2)
            if n1 < n2:
                return -1
            if n1 > n2:
                return 1
            return 0
        elif is_num1 and not is_num2:
            return -1   # numeric < alphanumeric
        elif not is_num1 and is_num2:
            return 1
        else:
            if id1 < id2:
                return -1
            if id1 > id2:
                return 1
            return 0

    major_a, minor_a, patch_a, pre_a = parse_version(a)
    major_b, minor_b, patch_b, pre_b = parse_version(b)

    # compare core versions
    if (major_a, minor_a, patch_a) < (major_b, minor_b, patch_b):
        return -1
    if (major_a, minor_a, patch_a) > (major_b, minor_b, patch_b):
        return 1

    # core equal
    if not pre_a and not pre_b:
        return 0
    if pre_a and not pre_b:
        return -1   # a has prerelease, b doesn't -> a lower
    if not pre_a and pre_b:
        return 1    # a no prerelease, b has -> a higher

    # both have prerelease: compare identifiers
    for id_a, id_b in zip(pre_a, pre_b):
        cmp = compare_ids(id_a, id_b)
        if cmp != 0:
            return cmp

    # all common identifiers equal; compare lengths
    if len(pre_a) < len(pre_b):
        return -1
    if len(pre_a) > len(pre_b):
        return 1
    return 0


REFERENCE_FUNC = compare_ref
REFERENCE_NAME = "compare_ref"

PROBE_INPUTS = [
    # major difference
    ("1.0.0", "2.0.0"),
    # minor difference
    ("1.0.0", "1.1.0"),
    # patch difference
    ("1.0.0", "1.0.1"),
    # equal
    ("1.0.0", "1.0.0"),
    # prerelease vs release
    ("1.0.0-alpha", "1.0.0"),
    # release vs prerelease
    ("1.0.0", "1.0.0-beta"),
    # both same prerelease
    ("1.0.0-alpha", "1.0.0-alpha"),
    # numeric vs alphanumeric id
    ("1.0.0-1", "1.0.0-alpha"),
    # alphanumeric comparison
    ("1.0.0-alpha", "1.0.0-beta"),
    # numeric comparison
    ("1.0.0-1", "1.0.0-2"),
    # leading zeros in numeric id (equal numerically)
    ("1.0.0-001", "1.0.0-1"),
    # more identifiers wins
    ("1.0.0-alpha.1", "1.0.0-alpha"),
    # fewer identifiers loses
    ("1.0.0-alpha", "1.0.0-alpha.1"),
    # build metadata ignored
    ("1.0.0+build", "1.0.0"),
    # build metadata with prerelease ignored
    ("1.0.0-alpha+build", "1.0.0-alpha"),
    # major difference with prerelease
    ("1.0.0-alpha", "2.0.0"),
    # mixed identifiers with more depth
    ("1.0.0-alpha.1", "1.0.0-alpha.beta"),
    # id containing hyphen
    ("1.0.0-alpha-1", "1.0.0-alpha-beta"),
    # multiple hyphens
    ("1.0.0-a-b", "1.0.0-a-c"),
    # all equal except extra id
    ("1.0.0-x.y.0", "1.0.0-x.y"),
]

EQUIV_KEY = lambda r: (r > 0) - (r < 0)


def check_impl(fn):
    def _expected_sign(a, b):
        def _parse(v):
            v = v.split('+')[0]
            if '-' in v:
                core_str, pre_str = v.split('-', 1)
            else:
                core_str, pre_str = v, ''
            parts = core_str.split('.')
            major = int(parts[0])
            minor = int(parts[1])
            patch = int(parts[2])
            pre_ids = pre_str.split('.') if pre_str else []
            return (major, minor, patch, pre_ids)

        def _cmp_ids(id1, id2):
            n1 = id1.isdigit()
            n2 = id2.isdigit()
            if n1 and n2:
                # numeric compare (int conversion handles leading zeros)
                diff = int(id1) - int(id2)
                if diff < 0: return -1
                if diff > 0: return 1
                return 0
            elif n1 and not n2:
                return -1
            elif not n1 and n2:
                return 1
            else:
                if id1 < id2: return -1
                if id1 > id2: return 1
                return 0

        maj_a, min_a, pat_a, pre_a = _parse(a)
        maj_b, min_b, pat_b, pre_b = _parse(b)

        if (maj_a, min_a, pat_a) < (maj_b, min_b, pat_b):
            return -1
        if (maj_a, min_a, pat_a) > (maj_b, min_b, pat_b):
            return 1

        if not pre_a and not pre_b:
            return 0
        if pre_a and not pre_b:
            return -1
        if not pre_a and pre_b:
            return 1

        # both have prerelease
        for id_a, id_b in zip(pre_a, pre_b):
            c = _cmp_ids(id_a, id_b)
            if c != 0:
                return c
        if len(pre_a) < len(pre_b):
            return -1
        if len(pre_a) > len(pre_b):
            return 1
        return 0

    results = []
    for pair in PROBE_INPUTS:
        a, b = pair
        expected = _expected_sign(a, b)
        actual = fn(pair)
        actual_sign = (actual > 0) - (actual < 0)
        results.append((a, b, actual_sign == expected))
    return results
